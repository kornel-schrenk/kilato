const fixturePlaces = (): Place[] => {
  const folder = join(import.meta.dirname, '../test/fixtures/site-data/cells')

const rad = (deg: number) => (deg * Math.PI) / 180
function metresBetween(a: { lat: number; lon: number }, b: { lat: number; lon: number }) {
  const x = Math.sin(rad(b.lat - a.lat) / 2) ** 2
  const y = Math.cos(rad(a.lat)) * Math.cos(rad(b.lat)) * Math.sin(rad(b.lon - a.lon) / 2) ** 2
  return 2 * 6371000 * Math.asin(Math.sqrt(x + y))
}

const specCategory: Record<CategoryId, number> = { sights: -200, notable: -100, leisure: -100, food: -50, services: 0, shops: 0 }

function statusAt(place: Place, at: number): 'open' | 'closed' | 'unknown' {
  if (!place.hours) return 'unknown'
  const minutes = (at - Date.parse(windowStart)) / 60000
  if (minutes < 0 || minutes >= 20160) return 'unknown'
  for (const [from, to] of place.hours) if (from <= minutes && minutes < to) return 'open'
  return 'closed'
}
const specStatus = { open: -100, closed: 300, unknown: 0 }
const collator = new Intl.Collator('hu')

/** Every expected row, from the spec by hand. */
function oracle(places: Place[], from: { lat: number; lon: number }, at: number, order: 'nearest' | 'relevant', off: CategoryId[] = []) {
  const rows = places
    .filter((place) => !off.includes(place.cat))
    .map((place) => ({ place, metres: metresBetween(from, place) }))
    .filter((row) => row.metres <= 1000)
    .map((row) => ({
      ...row,
      score: row.metres + (row.place.wiki ? -300 : 0) + specCategory[row.place.cat] + specStatus[statusAt(row.place, at)],
    }))
  if (order === 'nearest') rows.sort((a, b) => a.metres - b.metres)
  else rows.sort((a, b) => a.score - b.score || a.metres - b.metres || collator.compare(a.place.name, b.place.name))
  return rows.map((row) => row.place.name)
}

const rowNames = (page: Page) => page.locator('.place-name').allTextContents()

async function showAll(page: Page) {
  // Cells arrive one by one: with fewer than 100 rows, the button can still be on its way.
  await expect.poll(() => page.locator('.place').count()).toBeGreaterThanOrEqual(100)
  while (await button(page, 'Show more').count()) await button(page, 'Show more').click()
}

async function serve(page: Page, places: Place[], hoursWindowStart?: string) {
  const cells = new Map<string, Place[]>()
  for (const place of places) {
    const key = cellKey(place.lat, place.lon)
    cells.set(key, [...(cells.get(key) ?? []), place])
  }
  await page.route('**/kilato/data/manifest.json', (route) =>
    route.fulfill({ json: { cells: [...cells.keys()], wikiLangs: [], hoursWindowStart } }),
  )
  await page.route('**/kilato/data/cells/*.json', (route) =>
    route.fulfill({
      json: { places: cells.get(new URL(route.request().url()).pathname.split('/').pop()!.replace('.json', '')) ?? [] },
    }),
  )
}

const atOffset = (north: number, east = 0) => ({
  lat: deakTer.lat + (north / 111.19) * 0.001,
  lon: deakTer.lon + (east / 75.13) * 0.001,
})
const place = (id: string, north: number, more: Partial<Place> = {}): Place => ({
  id,
  cat: 'food',
  kind: 'cafe',
  name: id,
  ...atOffset(north),
  ...more,
})

const explore = async (page: Page) => {
  await page.goto('/kilato/')
  await button(page, 'Explore the map').click()
  await expect(page.locator('.pin')).toBeVisible()
  await expectNear(() => mapCentre(page), deakTer)
}

const times: [string, string][] = [
  ['Tue 11:00 local, most places open', '2026-11-03T10:00:00Z'],
  ['Tue 23:30 local, most places closed', '2026-11-03T22:30:00Z'],
  ['past the hours window, all unknown', '2026-12-01T10:00:00Z'],

for (const [label, time] of times) {
  test(`P1 fixture, ${label}: every row against the hand-typed table, each chip off, both orders`, async ({ page }) => {
    test.setTimeout(240_000)
    const at = Date.parse(time)
    const places = fixturePlaces()
    await page.clock.install({ time })
    await explore(page)
    await expect(page.locator('.place').first()).toBeVisible()
    await showAll(page)

    const nearest = oracle(places, deakTer, at, 'nearest')
    const relevant = oracle(places, deakTer, at, 'relevant')
    await expect.poll(() => rowNames(page)).toEqual(nearest)
    await button(page, 'Most relevant').click()
    await expect.poll(() => rowNames(page)).toEqual(relevant)
    const moved = relevant.filter((name, index) => nearest[index] !== name).length
    const statuses = { open: 0, closed: 0, unknown: 0 }
    for (const p of places) if (metresBetween(deakTer, p) <= 1000) statuses[statusAt(p, at)]++
    const wiki = places.filter((p) => p.wiki && metresBetween(deakTer, p) <= 1000).length
    log(`P1 ${label}: ${relevant.length} rows · ${moved} at a different position in Most relevant · open ${statuses.open} closed ${statuses.closed} unknown ${statuses.unknown} · with an article ${wiki} · both orders match, row by row`)

    const chips: [string, CategoryId][] = [
      ['Sights', 'sights'],
      ['Notable places', 'notable'],
      ['Food & drink', 'food'],
      ['Culture & leisure', 'leisure'],
      ['Services', 'services'],
      ['Shops', 'shops'],
    ]
    for (const [chip, id] of chips) {
      await button(page, chip).click()
      await expect.poll(() => rowNames(page)).toEqual(oracle(places, deakTer, at, 'relevant', [id]))
      await button(page, 'Nearest').click()
      await expect.poll(() => rowNames(page)).toEqual(oracle(places, deakTer, at, 'nearest', [id]))
      await button(page, 'Most relevant').click()
      await button(page, chip).click()
      await expect.poll(() => rowNames(page)).toEqual(relevant)
    }
    log(`P1 ${label}: each of the 6 chips off, in both orders, matches, and turning it on brings the rows back`)
  })
}

test('P6 2,000 places on this Raspberry Pi: how long a press takes', async ({ page }) => {
  test.setTimeout(120_000)
  const many = Array.from({ length: 2000 }, (_, i) => {
    const row = Math.floor(i / 50)
    const col = i % 50
    return place(`Place ${i}`, 0, {
      ...atOffset((row - 20) * 24, (col - 25) * 24),
      cat: (['food', 'shops', 'sights', 'services', 'notable', 'leisure'] as const)[i % 6],
      ...(i % 5 === 0 ? { wiki: { hu: `W${i}` } } : {}),
      ...(i % 3 === 0 ? { hours: i % 2 ? [] : [[0, 20160]] } : {}),
    } as Partial<Place>)
  })
  await page.clock.install({ time: '2026-11-03T10:00:00Z' })
  await serve(page, many, windowStart)
  await explore(page)
  await expect(page.locator('.place')).toHaveCount(100)
  const expected = oracle(many, deakTer, Date.parse('2026-11-03T10:00:00Z'), 'relevant').slice(0, 100)
  const press = (name: string) =>
    page.evaluate(async (label) => {
      const target = [...document.querySelectorAll('.order button')].find((b) => b.textContent === label) as HTMLButtonElement
      const start = performance.now()
      target.click()
      await new Promise((done) => queueMicrotask(() => done(null)))
      await new Promise((done) => setTimeout(done, 0))
      return performance.now() - start
    }, name)
  const timings: string[] = []
  for (const name of ['Most relevant', 'Nearest', 'Most relevant', 'Nearest', 'Most relevant']) {
    timings.push(`${name} ${(await press(name)).toFixed(0)} ms`)
  }
  await expect.poll(() => rowNames(page)).toEqual(expected)
  log(`P6 2,000 places within 1 km, 100 rows: ${timings.join(' · ')} · first 100 rows match the hand-typed table`)
})

test('P12 the toggle: sizes, contrast, narrow phone, wide screen', async ({ page }) => {
  const luminance = (rgb: string) => {
    const [r, g, b] = rgb.match(/\d+(\.\d+)?/g)!.slice(0, 3).map(Number).map((v) => {
      const c = v / 255
      return c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4
    })
    return 0.2126 * r + 0.7152 * g + 0.0722 * b
  }
  const ratio = (a: string, b: string) => {
    const [hi, lo] = [luminance(a), luminance(b)].sort((x, y) => y - x)
    return (hi + 0.05) / (lo + 0.05)
  }
  await page.setViewportSize({ width: 320, height: 568 })
  await page.goto('/kilato/')
  await button(page, 'HU').click()
  const narrow = await page.evaluate(() => {
    const on = document.querySelector('.order button[aria-pressed="true"]')!
    const off = document.querySelector('.order button[aria-pressed="false"]')!
    const read = (el: Element) => {
      const s = getComputedStyle(el)
      const r = el.getBoundingClientRect()
      return { color: s.color, background: s.backgroundColor, height: r.height, width: r.width, right: r.right, top: r.top, border: s.borderTopColor }
    }
    return { on: read(on), off: read(off), page: getComputedStyle(document.body).backgroundColor, scroll: document.documentElement.scrollWidth }
  })
  const pageBackground = narrow.page === 'rgba(0, 0, 0, 0)' ? 'rgb(255, 255, 255)' : narrow.page
  expect(narrow.scroll).toBe(320)
  expect(narrow.on.height).toBeGreaterThanOrEqual(44)
  expect(narrow.off.height).toBeGreaterThanOrEqual(44)
  log(`P12 320 px in Hungarian: no sideways scroll (${narrow.scroll}), buttons ${narrow.on.width.toFixed(0)}×${narrow.on.height.toFixed(0)} and ${narrow.off.width.toFixed(0)}×${narrow.off.height.toFixed(0)}, right edges ${narrow.on.right.toFixed(0)} and ${narrow.off.right.toFixed(0)}, ${narrow.on.top === narrow.off.top ? 'on one line' : 'on two lines'}`)
  log(`P12 contrast: chosen ${narrow.on.color} on ${narrow.on.background} = ${ratio(narrow.on.color, narrow.on.background).toFixed(2)}:1 · other ${narrow.off.color} on the page ${pageBackground} = ${ratio(narrow.off.color, pageBackground).toFixed(2)}:1 · border ${narrow.off.border} on the page = ${ratio(narrow.off.border, pageBackground).toFixed(2)}:1`)

  await page.setViewportSize({ width: 900, height: 700 })
  const wide = await page.evaluate(() => {
    const order = document.querySelector('.order')!.getBoundingClientRect()
    const map = document.querySelector('.map')!.getBoundingClientRect()
    const chips = document.querySelector('.chips')!.getBoundingClientRect()
    return { orderLeft: order.left, orderBottom: order.bottom, mapRight: map.right, chipsTop: chips.top }
  })
  expect(wide.orderLeft).toBeGreaterThanOrEqual(wide.mapRight)
  expect(wide.orderBottom).toBeLessThanOrEqual(wide.chipsTop)
  const { y } = await box(page, '.order')
  log(`P12 900 px: the toggle sits right of the map (left ${wide.orderLeft.toFixed(0)}, map ends ${wide.mapRight.toFixed(0)}), above the chips, y ${y.toFixed(0)}`)
})
