// QA's probe for #17. Own oracles: the spec's table read from the spec file, the colours typed in by hand,
// its own haversine over the fixture. It prints one PROBE line per fact. Copied into e2e/ to run, never committed there.
import { readdirSync, readFileSync } from 'node:fs'
import { join } from 'node:path'
import type { Page } from '@playwright/test'
import { cellKey } from '../src/core/cells'
import { expect, test } from './fixtures'
import { button, geography } from './map-helpers'

const say = (line: string) => console.log(`PROBE ${line}`)
const here = { lat: 47.4975, lon: 19.0545 }
const atOffset = (north: number, east = 0) => ({
  lat: here.lat + (north / 111.19) * 0.001,
  lon: here.lon + (east / 75.13) * 0.001,
})
// Typed by hand from the #14 marker table, not imported.
const colours: Record<string, string> = {
  sights: '#d55e00',
  notable: '#cc79a7',
  food: '#e69f00',
  leisure: '#009e73',
  services: '#56b4e9',
  shops: '#666666',
}
const ids = Object.keys(colours)

const specTable = () => {
  const spec = readFileSync(join(import.meta.dirname, '../spec/2026-11-02-kilato-mvp.md'), 'utf8')
  const section = spec.slice(spec.indexOf('### Category filter'), spec.indexOf('### Open now'))
  return [...section.matchAll(/^\| `(\w+)` \| (.+?) \| (.+?) \|$/gm)].map(([, id, en, hu]) => ({ id, en, hu }))
}
const enName = (id: string) => specTable().find((row) => row.id === id)!.en

const rad = (d: number) => (d * Math.PI) / 180
const hav = (a: { lat: number; lon: number }, b: { lat: number; lon: number }) => {
  const h =
    Math.sin(rad(b.lat - a.lat) / 2) ** 2 +
    Math.cos(rad(a.lat)) * Math.cos(rad(b.lat)) * Math.sin(rad(b.lon - a.lon) / 2) ** 2
  return 2 * 6371000 * Math.asin(Math.sqrt(h))
}
type P = { id: string; cat: string; kind: string; name: string; lat: number; lon: number }
const fixture = (): P[] => {
  const folder = join(import.meta.dirname, '../test/fixtures/site-data/cells')
  return readdirSync(folder).flatMap((file) => JSON.parse(readFileSync(join(folder, file), 'utf8')).places)
}

async function serve(page: Page, places: P[], { slowCells = 0 } = {}) {
  const cells = new Map<string, P[]>()
  for (const place of places) {
    const key = cellKey(place.lat, place.lon)
    cells.set(key, [...(cells.get(key) ?? []), place])
  }
  await page.route('**/kilato/data/manifest.json', (route) => route.fulfill({ json: { cells: [...cells.keys()] } }))
  await page.route('**/kilato/data/cells/*.json', async (route) => {
    if (slowCells) await new Promise((done) => setTimeout(done, slowCells))
    const key = new URL(route.request().url()).pathname.split('/').pop()!.replace('.json', '')
    await route.fulfill({ json: { places: cells.get(key) ?? [] } })
  })
}

const rowNames = (page: Page) => page.locator('.place-name').allTextContents()

async function explore(page: Page, name = 'Explore the map') {
  await page.goto('/kilato/')
  await button(page, name).click()
  await expect(page.locator('.pin')).toBeVisible()
  await expect(page.locator('img.leaflet-tile-loaded').first()).toBeAttached()
}

const pixel = (page: Page, point: { x: number; y: number }) =>
  page.evaluate(({ x, y }) => {
    const canvas = document.querySelector<HTMLCanvasElement>('.leaflet-places-pane canvas')
    if (!canvas) return null
    const area = canvas.getBoundingClientRect()
    const scale = canvas.width / area.width
    const [r, g, b, a] = canvas.getContext('2d')!.getImageData((x - area.x) * scale, (y - area.y) * scale, 1, 1).data
    return a === 0 ? null : `#${[r, g, b].map((part) => part.toString(16).padStart(2, '0')).join('')}`
  }, point)
const markerColor = async (page: Page, place: { lat: number; lon: number }) =>
  pixel(page, (await geography(page)).screen(place))

const place = (cat: string, north: number, east = 0): P => ({
  id: `${cat} place`,
  cat,
  kind: 'x',
  name: `${cat} place`,
  ...atOffset(north, east),
})
// Six places, one per category, far enough apart that no marker covers another.
const six = ids.map((cat, index) => place(cat, -250 + index * 100))

const trackErrors = (page: Page) => {
  const errors: string[] = []
  page.on('pageerror', (error) => errors.push(`pageerror: ${error.message}`))
  page.on('console', (message) => {
    if (message.type() === 'error') errors.push(`console: ${message.text()}`)
  })
  return errors
}

test('P1 six chips: the spec table, in order, in both languages, with the marker colours', async ({ page }) => {
  const errors = trackErrors(page)
  const table = specTable()
  expect(table.map((row) => row.id)).toEqual(ids)
  await page.goto('/kilato/')
  const chips = page.locator('.chip')
  await expect(chips).toHaveText(table.map((row) => row.en))
  say(`P1 EN chips == spec table: ${JSON.stringify(await chips.allTextContents())}`)
  await button(page, 'HU').click()
  await expect(chips).toHaveText(table.map((row) => row.hu))
  say(`P1 HU chips == spec table: ${JSON.stringify(await chips.allTextContents())}`)
  await button(page, 'EN').click()
  const read = await chips.evaluateAll((all) =>
    all.map((chip) => ({
      text: chip.textContent,
      pressed: chip.getAttribute('aria-pressed'),
      background: getComputedStyle(chip).backgroundColor,
      color: getComputedStyle(chip).color,
      height: chip.getBoundingClientRect().height,
    })),
  )
  for (const chip of read) say(`P1 ${JSON.stringify(chip)}`)
  const asRgb = (hex: string) => `rgb(${[1, 3, 5].map((i) => parseInt(hex.slice(i, i + 2), 16)).join(', ')})`
  expect(read.map((chip) => chip.background)).toEqual(ids.map((id) => asRgb(colours[id])))
  expect(read.every((chip) => chip.pressed === 'true')).toBe(true)
  expect(read.every((chip) => chip.height >= 44)).toBe(true)
  expect(errors).toEqual([])
})

test('P3 the full fixture: each chip off leaves exactly the list an own haversine predicts', async ({ page }) => {
  const errors = trackErrors(page)
  await explore(page)
  const near = fixture()
    .map((p) => ({ p, metres: hav(here, p) }))
    .filter(({ metres }) => metres <= 1000)
    .sort((a, b) => a.metres - b.metres)
  const expected = (off: string[]) => near.filter(({ p }) => !off.includes(p.cat)).map(({ p }) => p.name)
  await expect.poll(async () => page.locator('.place').count()).toBe(100)
  while (await button(page, 'Show more').count()) await button(page, 'Show more').click()
  await expect.poll(() => rowNames(page)).toHaveLength(near.length)
  say(`P3 fixture: ${near.length} places within 1 km, all listed`)
  for (const id of ids) {
    await button(page, enName(id)).click()
    await expect.poll(async () => (await rowNames(page)).sort()).toEqual(expected([id]).sort())
    const names = new Set(await rowNames(page))
    const hidden = near.filter(({ p }) => p.cat === id).length
    say(`P3 ${id} off: ${names.size} rows, ${hidden} hidden, matches the oracle`)
    await button(page, enName(id)).click()
    await expect.poll(async () => page.locator('.place').count()).toBe(near.length)
  }
  // And the other way: only one chip on.
  for (const id of ids) {
    for (const other of ids) if (other !== id) await button(page, enName(other)).click()
    const off = ids.filter((other) => other !== id)
    await expect.poll(async () => (await rowNames(page)).sort()).toEqual(expected(off).sort())
    say(`P3 only ${id} on: ${expected(off).length} rows, matches the oracle`)
    for (const other of ids) if (other !== id) await button(page, enName(other)).click()
    await expect.poll(async () => page.locator('.place').count()).toBe(near.length)
  }
  expect(errors).toEqual([])
})

test('P6 the filter holds when the point moves, when places arrive late, and sends nothing', async ({ page }) => {
  // One shop and one cafe near the start, one of each about 1.3 km north, in other cells.
  const near = [place('shops', 100), place('food', 150)]
  const far = [
    { ...place('shops', 1300), id: 'far shop', name: 'far shop' },
    { ...place('food', 1350), id: 'far cafe', name: 'far cafe' },
  ]
  await serve(page, [...near, ...far], { slowCells: 400 })
  const requests: string[] = []
  page.on('request', (request) => requests.push(request.url()))
  await page.goto('/kilato/')
  // Off before any place has loaded.
  await button(page, 'Shops').click()
  await button(page, 'Explore the map').click()
  await expect(page.locator('.pin')).toBeVisible()
  await expect.poll(() => rowNames(page)).toEqual(['food place'])
  say('P6 chip off before the point and the data: the places that arrive stay filtered')
  await expect.poll(() => markerColor(page, near[1])).toBe(colours.food)
  expect(await markerColor(page, near[0])).toBeNull()
  // Move the point north with a map tap.
  const target = atOffset(1000)
  const { x, y } = (await geography(page)).screen(target)
  await page.mouse.click(x, Math.max(y, 80))
  await expect.poll(async () => rowNames(page)).toContain('far cafe')
  const names = await rowNames(page)
  expect(names).not.toContain('far shop')
  await expect(button(page, 'Shops')).toHaveAttribute('aria-pressed', 'false')
  say(`P6 after the point moved: ${JSON.stringify(names)}; Shops stays off`)
  await button(page, 'Shops').click()
  await expect.poll(async () => rowNames(page)).toContain('far shop')
  say('P6 Shops back on: the far shop shows, with no reload')
  await expect.poll(async () => {
    await page.waitForTimeout(600)
    return requests.length
  }).toBe(requests.length)
  const count = requests.length
  for (let turn = 0; turn < 6; turn++) await button(page, 'Food & drink').click()
  await expect(button(page, 'Food & drink')).toHaveAttribute('aria-pressed', 'true')
  await page.waitForTimeout(500)
  say(`P6 six taps on a chip sent ${requests.length - count} requests`)
  expect(requests.length).toBe(count)
})

test('P12 readable: contrast of the name on and off, from the computed colours', async ({ page }) => {
  await page.goto('/kilato/')
  const parse = (css: string) => css.match(/\d+(\.\d+)?/g)!.slice(0, 3).map(Number)
  const lum = ([r, g, b]: number[]) => {
    const [x, y, z] = [r, g, b].map((v) => v / 255).map((v) => (v <= 0.03928 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4))
    return 0.2126 * x + 0.7152 * y + 0.0722 * z
  }
  const ratio = (a: number[], b: number[]) => {
    const [l, d] = [lum(a), lum(b)].sort((p, q) => q - p)
    return (l + 0.05) / (d + 0.05)
  }
  for (const id of ids) {
    const chip = button(page, enName(id))
    const on = await chip.evaluate((el) => [getComputedStyle(el).color, getComputedStyle(el).backgroundColor])
    const onRatio = ratio(parse(on[0]), parse(on[1]))
    await chip.click()
    const off = await chip.evaluate((el) => [getComputedStyle(el).color, getComputedStyle(el).borderTopColor])
    const white = [255, 255, 255]
    say(`P12 ${id}: on ${onRatio.toFixed(2)}:1 · off name on the page ${ratio(parse(off[0]), white).toFixed(2)}:1 · off border on the page ${ratio(parse(off[1]), white).toFixed(2)}:1`)
    expect(onRatio).toBeGreaterThanOrEqual(4.5)
    expect(ratio(parse(off[0]), white)).toBeGreaterThanOrEqual(4.5)
    await chip.click()
  }
})
