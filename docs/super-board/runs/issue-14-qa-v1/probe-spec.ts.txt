const R = 6371000
const rad = (d: number) => (d * Math.PI) / 180
const hav = (a: { lat: number; lon: number }, b: { lat: number; lon: number }) => {
  const h = Math.sin(rad(b.lat - a.lat) / 2) ** 2 + Math.cos(rad(a.lat)) * Math.cos(rad(b.lat)) * Math.sin(rad(b.lon - a.lon) / 2) ** 2
  return 2 * R * Math.asin(Math.sqrt(h))
}
const worldToScreen = (z: number, lat: number, lon: number) => {
  const size = 256 * 2 ** z
  const s = Math.sin(rad(lat))
  return { x: ((lon + 180) / 360) * size, y: (0.5 - Math.log((1 + s) / (1 - s)) / (4 * Math.PI)) * size }
}
/** Own reading of the view: a loaded tile of the current zoom gives zoom and origin. */
const screenOf = async (page: Page, p: { lat: number; lon: number }) => {
  const t = await page.evaluate(() => {
    const img = [...document.querySelectorAll<HTMLImageElement>('img.leaflet-tile-loaded')].find((i) => i.getBoundingClientRect().width === 256)!
    const [z, x, y] = new URL(img.src).pathname.slice(1).replace('.png', '').split('/').map(Number)
    const r = img.getBoundingClientRect()
    return { z, x, y, left: r.left, top: r.top }
  })
  const w = worldToScreen(t.z, p.lat, p.lon)
  return { x: w.x - t.x * 256 + t.left, y: w.y - t.y * 256 + t.top, z: t.z }
}
const pixelAt = (page: Page, x: number, y: number) =>
  page.evaluate(([x, y]) => {
    const c = document.querySelector<HTMLCanvasElement>('.leaflet-places-pane canvas')
    if (!c) return 'no-canvas'
    const a = c.getBoundingClientRect()
    const s = c.width / a.width
    const [r, g, b, al] = c.getContext('2d')!.getImageData((x - a.x) * s, (y - a.y) * s, 1, 1).data
    return al === 0 ? null : `#${[r, g, b].map((v) => v.toString(16).padStart(2, '0')).join('')}`
  }, [x, y])

async function stubTiles(page: Page) {
  await page.route('https://tile.openstreetmap.org/**', (route) => {
    const label = new URL(route.request().url()).pathname.slice(1).replace('.png', '')
    route.fulfill({
      contentType: 'image/svg+xml',
      body: `<svg xmlns="http://www.w3.org/2000/svg" width="256" height="256"><rect width="256" height="256" fill="#eef0ea"/><path d="M0 40H256M40 0V256" stroke="#dcdfd6" stroke-width="3"/><text x="128" y="128" text-anchor="middle" font-family="sans-serif" font-size="14" fill="#8a8d85">stub tile ${label}</text></svg>`,
    })
  })
}

async function explore(page: Page, label = 'Explore the map') {
  await page.goto('/kilato/')
  await page.getByRole('button', { name: label, exact: true }).click()
  await expect(page.locator('.pin')).toBeVisible()
  await expect(page.locator('.place').first()).toBeVisible()

}

test.describe('oracle', () => {
  for (const [lang, label, more] of [
    ['en', 'Explore the map', 'Show more'],
    ['hu', 'A térkép felfedezése', 'Továbbiak megjelenítése'],
  ] as const) {
    test.describe(lang, () => {
      test.use({ locale: lang === 'hu' ? 'hu-HU' : 'en-GB' })
      test(`every row matches QA's own haversine, in ${lang}`, async ({ page }) => {
        await explore(page, label)
        for (let i = 0; i < 3; i++) await page.getByRole('button', { name: more, exact: true }).click()
        await expect(page.locator('.place')).toHaveCount(oracle.length)
        const rows = await page.locator('.place').evaluateAll((els) =>
          els.map((el) => ({
            name: el.querySelector('.place-name')!.textContent,
            other: el.querySelector('.place-other')?.textContent ?? null,
            meta: el.querySelector('.place-meta')!.textContent!,
          })),
        )
        let bad = 0
        rows.forEach((row, i) => {
          const o = oracle[i]
          const metres = row.meta.split(' · ').pop()!
          const want = `${lang === 'en' ? o.rounded.toLocaleString('en-GB') : String(o.rounded)} m`
          const wantOther = o.names[lang] ?? null
          if (row.name !== o.name || metres !== want || row.other !== wantOther) {
            bad++
            log(`MISMATCH ${lang} row ${i}: ${JSON.stringify(row)} vs ${JSON.stringify({ name: o.name, want, wantOther })}`)
          }
        })
        log(`oracle ${lang}: ${rows.length} rows compared with QA's haversine (name, second name, rounded distance, order): ${bad} differ`)
        expect(bad).toBe(0)
        await expect(page.locator('.show-more')).toHaveCount(0)
      })
    })
  }
})

test.describe('privacy', () => {
  test.use({
    geolocation: { latitude: 47.49791, longitude: 19.05439, accuracy: 10 },
    permissions: ['geolocation'],
    viewport: { width: 375, height: 667 },
  })

  test('own capture: URLs, bodies, headers and storage over a whole visit', async ({ page, context }) => {
    const seen: { url: string; body: string; headers: string }[] = []
    context.on('request', async (r) => {
      seen.push({ url: r.url(), body: r.postData() ?? '', headers: JSON.stringify(r.headers()) })
    })
    await page.goto('/kilato/')
    await expect(page.locator('.location-dot')).toBeVisible()
    await expect(page.locator('.place')).toHaveCount(100)
    await page.locator('.place').nth(1).click()
    await page.getByRole('button', { name: 'Show more', exact: true }).click()
    await expect(page.locator('.place')).toHaveCount(200)
    // a walk, a drag, a tap, then back
    await context.setGeolocation({ latitude: 47.5005, longitude: 19.0577, accuracy: 10 })
    await page.waitForTimeout(600)
    const box = (await page.locator('.leaflet-container').boundingBox())!
    await page.mouse.move(box.x + 180, box.y + 150)
    await page.mouse.down()
    await page.mouse.move(box.x + 120, box.y + 90, { steps: 10 })
    await page.mouse.up()
    await page.waitForTimeout(800)
    await page.mouse.click(box.x + 60, box.y + 60)
    await page.waitForTimeout(500)
    const needles = ['47.497', '19.054', '47.49', '19.05', '47.50', '4749791', '1905439']
    const hits = seen.filter((s) => needles.some((n) => s.url.includes(n) || s.body.includes(n) || s.headers.includes(n)))
    const hosts = [...new Set(seen.map((s) => new URL(s.url).host.replace(/^127\.0\.0\.1:\d+$/, 'preview')))]
    const storage = await page.evaluate(() => ({ local: JSON.stringify(localStorage), session: JSON.stringify(sessionStorage), cookie: document.cookie }))
    log(`privacy: ${seen.length} requests to ${hosts.join(', ')}; ${seen.filter((s) => s.body).length} with a body; ${hits.length} holding any of ${needles.join(' ')} in URL, body or headers; storage ${JSON.stringify(storage)}`)
    for (const h of hits) log(`  HIT ${h.url}`)
    log(`cell requests: ${seen.filter((s) => s.url.includes('/cells/')).map((s) => new URL(s.url).pathname.split('/').pop()).join(' ')}`)
    expect(hits).toEqual([])
  })
})

        const r = c.getBoundingClientRect()

        const opts = { bubbles: true, clientX: r.x + 40, clientY: r.y + 40, button: 0 }
        return { started: t, x: opts.clientX, y: opts.clientY }
      })
      const t2 = Date.now()
      await page.mouse.click(move.x, move.y)
      await page.waitForFunction(() => document.querySelector('.place') !== null)
      await page.waitForTimeout(0)
      const moved = Date.now() - t2
      log(`volume ${count}: Explore to first rows ${firstRows} ms, row tap to popup ${rowTap} ms, tap on the map back to a list ${moved} ms (Pi, 4 cores, other lanes may be running)`)
      await expect(page.locator('.place').first()).toBeVisible()
    })
  }
})

test.describe('more edges', () => {
  test.use({ viewport: { width: 1280, height: 720 } })

  test('a double click on a marker selects it and does not zoom; beside a marker it zooms', async ({ page }) => {
    await explore(page)
    const lonely = oracle.find((c) => oracle.every((x) => x === c || hav(x, c) > 40) && hav(c, deak) < 500)!
    const s = await screenOf(page, lonely)
    await page.mouse.dblclick(s.x, s.y)
    await page.waitForTimeout(700)
    const after = await screenOf(page, deak)
    log(`double click on a marker: zoom ${s.z} then ${after.z}, highlighted "${await page.locator('[aria-current] .place-name').textContent()}"`)
    await expect(page.locator('[aria-current]')).toHaveCount(1)
  })

  test('a row tapped while the map coasts after a flick', async ({ page }) => {
    await explore(page)
    const box = (await page.locator('.leaflet-container').boundingBox())!
    const x = box.x + box.width / 2, y = box.y + box.height / 2
    await page.mouse.move(x, y)
    await page.mouse.down()
    for (let i = 1; i <= 6; i++) { await page.mouse.move(x - i * 30, y); await page.waitForTimeout(8) }
    await page.mouse.up()
    await page.locator('.place').first().click()
    await page.waitForTimeout(900)
    await expect(page.locator('[aria-current]')).toHaveCount(1)
    const pin = (await page.locator('.pin').boundingBox())!
    const mapBox = (await page.locator('.leaflet-container').boundingBox())!
    log(`row tap during a coast: pin tip at ${Math.round(pin.x + pin.width / 2 - mapBox.x)}, ${Math.round(pin.y + pin.height - mapBox.y)} on a map ${mapBox.width} x ${mapBox.height}; popup count ${await page.locator('.leaflet-popup-content').count()}; ${await page.locator('.place').count()} rows`)
    await expect(page.locator('.place').first()).toBeVisible()
  })

  test('a language switch keeps the highlight, the popup and the markers', async ({ page }) => {
    await explore(page)
    await page.locator('.place').nth(1).click()
    await expect(page.locator('.leaflet-popup-content')).toBeVisible()
    await page.getByRole('button', { name: 'HU', exact: true }).click()
    await expect(page.locator('.place').nth(1)).toHaveAttribute('aria-current', 'true')
    await expect(page.locator('.leaflet-popup-content')).toBeVisible()
    await expect(page.locator('.place-meta').first()).toContainText('Wikipédia-szócikk')
    const s = await screenOf(page, oracle[1])
    log(`language switch: row and popup stay, ring pixel ${await pixelAt(page, s.x, s.y - 8)}`)
    expect(await pixelAt(page, s.x, s.y - 8)).toBe('#000000')
  })

  test('zero places: an empty manifest gives no rows, no markers, no failure', async ({ page }) => {
    await page.route('**/kilato/data/manifest.json', (r) => r.fulfill({ json: { cells: [] } }))
    await page.goto('/kilato/')
    await page.getByRole('button', { name: 'Explore the map', exact: true }).click()
    await expect(page.locator('.pin')).toBeVisible()
    await page.waitForTimeout(400)
    log(`empty manifest: ${await page.locator('.place').count()} rows, ${await page.locator('.show-more').count()} Show more, ${await page.locator('.load-error').count()} failure messages`)
    await expect(page.locator('.load-error')).toHaveCount(0)
  })

  for (const [what, body] of [['a manifest with no cells field', '{}'], ['a cell with no places field', '{}']] as const) {
    test(`malformed data: ${what}`, async ({ page }) => {
      const errors: string[] = []
      page.on('pageerror', (e) => errors.push(e.message))
      if (what.startsWith('a manifest')) {
        await page.route('**/kilato/data/manifest.json', (r) => r.fulfill({ contentType: 'application/json', body }))
      } else {
        await page.route('**/kilato/data/cells/4749_1270.json', (r) => r.fulfill({ contentType: 'application/json', body }))
      }
      await page.goto('/kilato/')
      await page.getByRole('button', { name: 'Explore the map', exact: true }).click()
      await expect(page.locator('.pin')).toBeVisible()
      await page.waitForTimeout(500)
      const first = await page.locator('.load-error').count()
      let afterRetry = 'no Retry to press'
      if (first) {
        await page.locator('.load-error button').click()
        await page.waitForTimeout(500)
        afterRetry = `${await page.locator('.load-error').count()} messages after Retry, ${await page.locator('.place').count()} rows`
      }
      log(`malformed, ${what}: ${first} message, ${afterRetry}, page errors ${JSON.stringify(errors)}`)
      await expect(page.locator('.pin')).toBeVisible()
    })
  }
})
